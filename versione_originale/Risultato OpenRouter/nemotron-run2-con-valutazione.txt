CREATE TEMP TABLE tasso_cambio AS
SELECT 'EUR' AS Valuta, 1.08 AS Tasso UNION ALL
SELECT 'GBP', 1.25 UNION ALL
SELECT 'JPY', 975.0/140000.0;

CREATE TEMP TABLE vendite_usd AS
SELECT VENDITE_PC.Modello, VENDITE_PC.Prezzo_Locale * tasso_cambio.Tasso AS Prezzo_USD
FROM VENDITE_PC
JOIN tasso_cambio ON VENDITE_PC.Valuta_Locale = tasso_cambio.Valuta;

CREATE TEMP TABLE media_usd AS
SELECT Modello AS Modello_PC, AVG(Prezzo_USD) AS Prezzo_Medio_USD
FROM vendite_usd
GROUP BY Modello;
